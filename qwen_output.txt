# backend/app/tools/business_rules.py

def tinh_size(can_nang: float, chieu_cao: float, loai_san_pham: str = "cả hai") -> dict:
    # --- Size Áo ---
    if can_nang < 62:
        ao = "S"
    elif can_nang <= 75:
        ao = "M"
    elif can_nang <= 84:
        ao = "L"
    else:
        ao = "XL"

    # Upsize nếu chiều cao < 1.65
    if chieu_cao < 1.65:
        upsize_map = {"S": "M", "M": "L", "L": "XL", "XL": "XXL"}
        ao = upsize_map[ao]

    # --- Size Quần ---
    if can_nang <= 70:
        quan = "M"
    elif can_nang <= 85:
        quan = "L"
    else:
        quan = "XL"

    # --- Trả kết quả theo loại sản phẩm ---
    if loai_san_pham == "áo":
        return {"size": f"Áo {ao}", "can_nang": can_nang, "chieu_cao": chieu_cao}
    elif loai_san_pham == "quần":
        return {"size": f"Quần {quan}", "can_nang": can_nang, "chieu_cao": chieu_cao}
    else:
        return {"size": f"Áo {ao}, Quần {quan}", "can_nang": can_nang, "chieu_cao": chieu_cao}
