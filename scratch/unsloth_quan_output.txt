import re

def chuan_hoa_ma_quan(text: str) -> str:
    # Định nghĩa các từ khóa chỉ quần
    keywords = r'\b(?:quần|quan|q|short|đùi)\b'
    
    def replace_match(match):
        full_match = match.group(0)
        prefix = match.group(1)
        numbers = match.group(2)
        
        # Kiểm tra nếu chuỗi số hợp lệ (ít nhất 2 ký tự, chỉ chứa 1,2,3,4,6,7)
        if len(numbers) >= 2 and all(c in '123467' for c in numbers):
            # Tách các chữ số và nối bằng dấu phẩy
            formatted = ', '.join(list(numbers))
            return f"{prefix}{formatted}"
        else:
            return full_match  # Không thay đổi nếu không hợp lệ

    # Tìm tất cả các chuỗi phù hợp: từ khóa + chuỗi số liên tiếp
    pattern = rf'({keywords})\s*(\d+)'
    result = re.sub(pattern, replace_match, text, flags=re.IGNORECASE)
    
    return result
